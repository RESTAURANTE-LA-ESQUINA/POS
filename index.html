Esquema SQL para Supabase

Ejecuta esto en el SQL Editor de tu panel de Supabase para inicializar las tablas y políticas de seguridad.

-- 1. CREACIÓN DE TABLAS
CREATE TABLE IF NOT EXISTS mesas (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  numero INT NOT NULL UNIQUE,
  capacidad INT DEFAULT 4,
  estado TEXT DEFAULT 'disponible' CHECK (estado IN ('disponible', 'ocupada', 'reservada')),
  cliente TEXT DEFAULT NULL
);

CREATE TABLE IF NOT EXISTS productos (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  nombre TEXT NOT NULL,
  categoria TEXT NOT NULL,
  estacion TEXT CHECK (estacion IN ('cocina', 'barra')),
  precio NUMERIC(10,2) NOT NULL
);

CREATE TABLE IF NOT EXISTS pedidos (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  origen TEXT NOT NULL,
  tipo TEXT CHECK (tipo IN ('mesa', 'para-llevar', 'domicilio', 'venta')),
  estado TEXT DEFAULT 'pendiente' CHECK (estado IN ('pendiente', 'preparacion', 'listo', 'pagado', 'cancelado')),
  metodo_pago TEXT DEFAULT NULL,
  subtotal NUMERIC(10,2) DEFAULT 0,
  impuesto NUMERIC(10,2) DEFAULT 0,
  total NUMERIC(10,2) DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS detalles_pedido (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  pedido_id BIGINT REFERENCES pedidos(id) ON DELETE CASCADE,
  producto_id BIGINT REFERENCES productos(id),
  nombre_producto TEXT NOT NULL,
  cantidad INT NOT NULL,
  precio_unitario NUMERIC(10,2) NOT NULL,
  notas TEXT DEFAULT '',
  estacion TEXT CHECK (estacion IN ('cocina', 'barra'))
);

CREATE TABLE IF NOT EXISTS inventario (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  insumo TEXT NOT NULL,
  unidad TEXT NOT NULL,
  stock_actual NUMERIC(10,2) DEFAULT 0,
  stock_minimo NUMERIC(10,2) DEFAULT 5
);

CREATE TABLE IF NOT EXISTS personal_usuarios (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  username TEXT UNIQUE NOT NULL,
  email TEXT UNIQUE NOT NULL,
  rol TEXT CHECK (rol IN ('administrador', 'cajero', 'mesero', 'cocinero')),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- 2. HABILITAR RLS Y POLITICAS PUBLICAS (PERMITIR ACCESO CON ANON KEY)
ALTER TABLE mesas ENABLE ROW LEVEL SECURITY;
ALTER TABLE productos ENABLE ROW LEVEL SECURITY;
ALTER TABLE pedidos ENABLE ROW LEVEL SECURITY;
ALTER TABLE detalles_pedido ENABLE ROW LEVEL SECURITY;
ALTER TABLE inventario ENABLE ROW LEVEL SECURITY;
ALTER TABLE personal_usuarios ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Mesas Public" ON mesas FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Productos Public" ON productos FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Pedidos Public" ON pedidos FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Detalles Public" ON detalles_pedido FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Inventario Public" ON inventario FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "Personal Public" ON personal_usuarios FOR ALL USING (true) WITH CHECK (true);
